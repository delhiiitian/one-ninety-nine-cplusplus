cmake_minimum_required(VERSION 3.16)
project(LogAnalyzer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(log_analyzer src/main.cpp src/analyzer.cpp src/stats.cpp src/log_entry.cpp)
target_compile_options(log_analyzer PRIVATE -Wall -Wextra -pedantic)
target_link_libraries(log_analyzer PRIVATE Threads::Threads)
